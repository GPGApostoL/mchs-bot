USER_SESSIONS = {}

def create_gold_progress_bar(percent: int) -> str:
    """Визуальный прогресс-бар в премиум стиле"""
    filled = int(10 * percent // 100)
    return '🟨' * filled + '⬛' * (10 - filled)

async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    await start_new_test(user_id, context, update)

async def start_new_test(user_id: int, context: ContextTypes.DEFAULT_TYPE, update: Update = None):
    # Генерируем выборку РОВНО из 50 вопросов (или максимум доступных)
    total_count = min(50, len(QUIZ_QUESTIONS))
    selected_questions = random.sample(QUIZ_QUESTIONS, total_count)
    
    USER_SESSIONS[user_id] = {
        "questions": selected_questions,
        "current_index": 0,
        "score": 0,
        "total": total_count
    }
    
    start_message = (
        "💎 **КВАЛИФИКАЦИОННОЕ ТЕСТИРОВАНИЕ**\n"
        "🏛 *Министерство по чрезвычайным ситуациям*\n"
        "━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n"
        "📋 **Сведения о сессии:**\n"
        f"▪️ **Объем выборки:** `{total_count}` вопросов\n"
        "▪️ **Формат:** Викторина (Instant Feedback)\n"
        "▪️ **Статус:** Активен\n\n"
        "⚠️ *Выберите вариант ответа. Система зафиксирует ваш выбор и покажет верный вариант.*"
    )
    
    if update and update.message:
        await update.message.reply_text(start_message, parse_mode="Markdown")
    else:
        await context.bot.send_message(chat_id=user_id, text=start_message, parse_mode="Markdown")
        
    await send_next_question(context, user_id)

async def send_next_question(context: ContextTypes.DEFAULT_TYPE, user_id: int):
    session = USER_SESSIONS.get(user_id)
    if not session:
        return

    idx = session["current_index"]
    
    # Завершение теста при достижении конца выборки
    if idx >= session["total"]:
        await show_results(context, user_id)
        return

    qdata = session["questions"][idx]
    
    # Перемешиваем ответы
    options = qdata["options"].copy()
    correct_text = options[qdata["correct"]]
    random.shuffle(options)
    correct_id = options.index(correct_text)

    session["current_correct_id"] = correct_id
    question_text = f"⚖️ ВОПРОС [{idx + 1}/{session['total']}]\n\n{qdata['question']}"

    message = await context.bot.send_poll(
        chat_id=user_id,
        question=question_text,
        options=options,
        type=Poll.QUIZ,
        correct_option_id=correct_id,
        is_anonymous=False
    )
    
    context.bot_data[message.poll.id] = {
        "user_id": user_id,
        "correct_option_id": correct_id
    }

async def receive_poll_answer(update: Update, context: ContextTypes.DEFAULT_TYPE):
    answer = update.poll_answer
    poll_id = answer.poll_id
    poll_data = context.bot_data.get(poll_id)
    
    if not poll_data:
        return

    user_id = poll_data["user_id"]
    session = USER_SESSIONS.get(user_id)
    if not session:
        return

    # Учитываем балл если ответили верно
    selected_option = answer.option_ids[0]
    if selected_option == poll_data["correct_option_id"]:
        session["score"] += 1

    # Переходим к следующему вопросу
    session["current_index"] += 1
    await send_next_question(context, user_id)

async def show_results(context: ContextTypes.DEFAULT_TYPE, user_id: int):
    session = USER_SESSIONS.get(user_id)
    if not session:
        return

    score = session["score"]
    total = session["total"]
    wrong = total - score
    percent = round((score / total) * 100)
    
    if percent >= 90:
        badge = "👑 **ВЫСШИЙ БАЛЛ / ЭКСПЕРТ**"
        status_note = "Ваша теоретическая подготовка находится на безупречном уровне."
    elif percent >= 75:
        badge = "🥇 **АТТЕСТАЦИЯ ПРОЙДЕНА**"
        status_note = "Вы продемонстрировали высокий уровень профессиональных знаний."
    elif percent >= 50:
        badge = "🥈 **УДОВЛЕТВОРИТЕЛЬНО**"
        status_note = "Тест сдан, однако рекомендуется повторить отдельные разделы."
    else:
        badge = "🛑 **АТТЕСТАЦИЯ НЕ ПРОЙДЕНА**"
        status_note = "Необходимо детально изучить нормативно-правовую базу."

    progress_bar = create_gold_progress_bar(percent)

    result_card = (
        "🏆 **ОФИЦИАЛЬНЫЙ ПРОТОКОЛ ТЕСТИРОВАНИЯ**\n"
        "━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n\n"
        f"Квалификация: {badge}\n\n"
        f"Прогресс: {progress_bar} `{percent}%`\n\n"
        "📊 **Детализация результатов:**\n"
        f" ▫️ Правильных ответов: `{score}`\n"
        f" ▫️ Ошибочных ответов: `{wrong}`\n"
        f" ▫️ Всего обработано: `{total}`\n\n"
        f"📝 *{status_note}*\n"
        "━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    )

    keyboard = [
        [
            InlineKeyboardButton("🔄 Начать новый тест", callback_data="restart_quiz"),
            InlineKeyboardButton("Завершить", callback_data="finish_quiz")
        ]
    ]
    reply_markup = InlineKeyboardMarkup(keyboard)

    await context.bot.send_message(
        chat_id=user_id,
        text=result_card,
        reply_markup=reply_markup,
        parse_mode="Markdown"
    )

async def button_click(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    user_id = query.from_user.id

    if query.data == "restart_quiz":
        await query.edit_message_text("⏳ *Формирование новой сессии на 50 вопросов...*", parse_mode="Markdown")
        await start_new_test(user_id, context)
    elif query.data == "finish_quiz":
        if user_id in USER_SESSIONS:
            del USER_SESSIONS[user_id]
        await query.edit_message_text(
            "🔒 **Сессия завершена.**\n\nДля повторного тестирования воспользуйтесь командой /start.",
            parse_mode="Markdown"
        )

def main():
    app = Application.builder().token(TOKEN).build()
    app.add_handler(CommandHandler("start", start))
    app.add_handler(PollAnswerHandler(receive_poll_answer))
    app.add_handler(CallbackQueryHandler(button_click))
    print("⚜️ Премиум-бот с 50 вопросами запущен!")
    app.run_polling()

if __name__ == "__main__":
    main()
